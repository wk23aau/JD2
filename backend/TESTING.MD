# Backend API Manual Testing Checklist

This document provides a checklist for manually testing the backend API of the AI CV Maker application.

## I. Prerequisites & Tools

*   **Backend Server Running:** Ensure the backend server is running locally or on a staging environment.
*   **Database Access:** Access to the application database is required to verify data creation, updates, and deletions.
*   **API Testing Tool:** Use a tool like Postman, Insomnia, or `curl` to make HTTP requests to the API endpoints.
*   **Valid JWT Token:** For testing protected endpoints, obtain a JWT token by logging in or registering a test user.
*   **Google Account(s):** At least one, preferably two, Google accounts for testing OAuth flows.
*   **Google Cloud Project:** Ensure OAuth credentials and Gemini API key are correctly configured in the backend's `.env` file.

## II. Test Case Format

For each test case, use the following format:

*   **Test ID:** A unique identifier (e.g., AUTH-001).
*   **Description:** A brief summary of the test objective.
*   **Steps to Reproduce:** Detailed steps to execute the test.
*   **Expected Results:** The anticipated outcome, including HTTP status codes and key aspects of the response body.
*   **Actual Results:** (To be filled by the tester)
*   **Pass/Fail:** (To be filled by the tester)
*   **Notes:** Any additional comments or observations.

import org.apache.hadoop.fs.Path;
import org.apache.hadoop.io.Text;
import org.apache.hadoop.mapreduce.lib.input.FileInputFormat;
import org.apache.hadoop.mapreduce.lib.output.TextOutputFormat;
import org.apache.hadoop.mapreduce.Job;
import org.apache.hadoop.conf.Configuration;
import org.apache.hadoop.mapreduce.Mapper;
import org.apache.hadoop.mapreduce.Reducer;
import org.apache.hadoop.util.GenericOptionsParser;
import java.io.IOException;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

public class Task1 {
    public static class TokenizerMapper extends Mapper<Object, Text, Text, Text> {
        private Text word = new Text();
        private Text documentId = new Text();

        @Override
        public void map(Object key, Text value, Context context) throws IOException, InterruptedException {
            String line = value.toString();
            // Extract document ID using regex, assuming document IDs are in the format "ID: <id>"
            Pattern pattern = Pattern.compile("ID: (\\w+)");
            Matcher matcher = pattern.matcher(line);
            if (matcher.find()) {
                documentId.set(matcher.group(1));
            }

            String[] words = line.split("\\s+");
            for (String token : words) {
                // Remove non-alphanumeric characters and convert to lowercase
                String cleanedToken = token.replaceAll("[^a-zA-Z0-9]", "").toLowerCase();
                if (!cleanedToken.isEmpty()) {
                    word.set(cleanedToken);
                    context.write(word, documentId);
                }
            }
        }
    }

    public static class IntSumReducer extends Reducer<Text, Text, Text, Text> {
        private Text result = new Text();

        @Override
        public void reduce(Text key, Iterable<Text> values, Context context) throws IOException, InterruptedException {
            // Build a comma-separated string of unique document IDs
            StringBuilder docIds = new StringBuilder();
            for (Text val : values) {
                if (docIds.length() > 0) {
                    docIds.append(",");
                }
                docIds.append(val.toString());
            }
            result.set(docIds.toString());
            context.write(key, result);
        }
    }

    public static void main(String[] args) throws Exception {
        Configuration conf = new Configuration();
        String[] otherArgs = new GenericOptionsParser(conf, args).getRemainingArgs();
        if (otherArgs.length != 2) {
            System.err.println("Usage: Task1 <in> <out>");
            System.exit(2);
        }
        Job job = Job.getInstance(conf, "Task1");
        job.setJarByClass(Task1.class);
        job.setMapperClass(TokenizerMapper.class);
        job.setCombinerClass(IntSumReducer.class);
        job.setReducerClass(IntSumReducer.class);
        job.setOutputKeyClass(Text.class);
        job.setOutputValueClass(Text.class);
        FileInputFormat.addInputPath(job, new Path(otherArgs[0]));
        TextOutputFormat.setOutputPath(job, new Path(otherArgs[1]));
        System.exit(job.waitForCompletion(true) ? 0 : 1);
    }
}
